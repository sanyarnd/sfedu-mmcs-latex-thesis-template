\section{Код и псевдокод}
\LaTeX{} позволяет вставлять как исходный код (пакет \texttt{listings}),
так и псевдокод (пакет \texttt{algpseudocode}). Оформление обоих окружений
настраивается в \texttt{commands.tex}.

\begin{lstlisting}[caption={Вызов БПФ в библиотеке \texttt{cuFFT}}, label={lst:cufft}]
cufftComplex *d_signal;
cudaMalloc((void **) &d_signal, mem_size);
cudaMemcpy(d_signal, fg, mem_size, cudaMemcpyHostToDevice);

cufftHandle plan;
cufftPlan2d(&plan, N, N, CUFFT_C2C);

// прямое преобразование
cufftExecC2C(plan, d_signal, d_signal, CUFFT_FORWARD);
\end{lstlisting}

Код можно подключать и из отдельного файла:
\begin{center}
    \verb+\lstinputlisting[language=Python]{code/main.py}+
\end{center}

Ключевые слова псевдокода можно переводить (см. \texttt{commands.tex}),
см. \autoref{alg:example}.
\begin{algorithm}[H]
    \caption{Пример псевдокода}\label{alg:example}
    \begin{algorithmic}[1]
        \Procedure{F}{$A$, $B$, $N$}
            \State $E \gets A$
            \For{$i \gets 1$ \textbf{to} $N$}
                \State $\hat{E} \gets \Call{FFT}{E}$
            \EndFor
            \State \Return $E$
        \EndProcedure
    \end{algorithmic}
\end{algorithm}
